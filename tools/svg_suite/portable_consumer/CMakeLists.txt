cmake_minimum_required(VERSION 3.24)
project(svg_suite_portable_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." suite)

function(check_portable_closure target)
  get_property(visited GLOBAL PROPERTY SVG_SUITE_CHECKED_TARGETS)
  if(NOT TARGET "${target}" OR "${target}" IN_LIST visited)
    return()
  endif()
  set_property(GLOBAL APPEND PROPERTY SVG_SUITE_CHECKED_TARGETS "${target}")
  get_target_property(sources "${target}" SOURCES)
  foreach(source IN LISTS sources)
    if(source MATCHES "(^|/)(Provenance|ProcessRenderer)\\.cc$")
      message(FATAL_ERROR "Portable callback target depends on POSIX source: ${source}")
    endif()
  endforeach()
  get_target_property(dependencies "${target}" LINK_LIBRARIES)
  foreach(dependency IN LISTS dependencies)
    check_portable_closure("${dependency}")
  endforeach()
endfunction()

# Check the callback dependency closure even on hosts that have POSIX headers.
check_portable_closure(svg_suite)
add_executable(callback-consumer CallbackConsumer.cc)
target_link_libraries(callback-consumer PRIVATE svg_suite)
add_test(NAME callback-consumer COMMAND callback-consumer)
